class Solution:
    def sortArray(self, nums: List[int]) -> List[int]:
        def mergeSort(arr):
            if len(arr) <= 1:
                return arr
            mid = len(arr) // 2
            left = mergeSort(arr[:mid])
            right = mergeSort(arr[mid:])
            i, j, res = 0, 0, []
            while i < len(left) and j < len(right):
                if left[i] < right[j]:
                    res.append(left[i])
                    i += 1
                else:
                    res.append(right[j])
                    j += 1
            res.extend(left[i:])
            res.extend(right[j:])
            return res
        return mergeSort(nums)

        Do not return anything, modify nums in-place instead.

        high = len(nums) - 1

        while i <= high:

                i += 1

                high -= 1

                i += 1

from collections import deque
class MovingAverage:

    def __init__(self, size: int):
        self.cur_size = 0
        self.max_size = size
        self.holder = deque(maxlen=size)
        self.cur_sum = 0

    def next(self, val: int) -> float:        
        self.cur_size += 1
        divisor = min(self.cur_size, self.max_size)
        if len(self.holder) == self.max_size:
            evicted = self.holder[0]
            self.cur_sum -= evicted            
        self.holder.append(val)
        self.cur_sum += val
        return self.cur_sum / divisor

class Solution:
    def findDuplicate(self, paths: list[str]) -> list[list[str]]:
        content_path_dict = {}
        for i in range(len(paths)):
            contents = paths[i].split(" ")
            file_path = contents[0]
            for j in range(1, len(contents)):
                file_string = contents[j].split("(")
                file_content = file_string[1].rstrip(")")
                file_name = file_string[0]
                complete_path = file_path + "/" + file_name
                if file_content not in content_path_dict:
                    content_path_dict[file_content] = [complete_path]
                else:
                    content_path_dict[file_content].append(complete_path)

        res = []
        for key, value in content_path_dict.items():
            if len(value) > 1:
                res.append(value)
        return res
